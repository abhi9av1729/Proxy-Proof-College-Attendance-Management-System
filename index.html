function doPost(e) {
  try {
    var data = {};
    if (e && e.postData && e.postData.contents) {
      try {
        data = JSON.parse(e.postData.contents);
      } catch (err) {
        data = e.parameter || {};
      }
    } else if (e && e.parameter) {
      data = e.parameter;
    }
    
    // Global Clear Logs signal across all devices
    if (data.action === "clearLogs") {
      var props = PropertiesService.getScriptProperties();
      props.setProperty("LAST_CLEARED_AT", String(data.timestamp || Date.now()));
      return ContentService.createTextOutput(JSON.stringify({result: "success"}))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    // Save ALL Config & Credentials globally across all devices
    if (data.action === "saveConfig") {
      var props = PropertiesService.getScriptProperties();
      if (data.webAppUrl) props.setProperty("WEBAPP_URL", String(data.webAppUrl));
      if (data.repPasscode) props.setProperty("REP_PASSCODE", String(data.repPasscode));
      if (data.studentPasscode) props.setProperty("STUDENT_PASSCODE", String(data.studentPasscode));
      if (data.adminUser) props.setProperty("ADMIN_USER", String(data.adminUser));
      if (data.adminPass) props.setProperty("ADMIN_PASS", String(data.adminPass));
      return ContentService.createTextOutput(JSON.stringify({result: "success"}))
        .setMimeType(ContentService.MimeType.JSON);
    }
    
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) {
      return ContentService.createTextOutput(JSON.stringify({
        result: "error", 
        message: "Spreadsheet not found."
      })).setMimeType(ContentService.MimeType.JSON);
    }
    
    var sheet = ss.getSheets()[0];
    var now = new Date();
    var dateStr = data.dateStr || Utilities.formatDate(now, Session.getScriptTimeZone(), "yyyy-MM-dd");
    var timeStr = data.timeStr || Utilities.formatDate(now, Session.getScriptTimeZone(), "HH:mm:ss");
    
    sheet.appendRow([
      dateStr,
      timeStr,
      data.regNo || "N/A",
      data.name || "N/A",
      data.course || "N/A",
      data.section || "N/A"
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({result: "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({result: "error", message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

// Enables real-time cross-device attendance, passcodes & settings sync across all devices
function doGet(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheets()[0];
    var data = sheet.getDataRange().getValues();
    
    var props = PropertiesService.getScriptProperties();
    var lastClearedAt = props.getProperty("LAST_CLEARED_AT") || "0";
    var webAppUrl = props.getProperty("WEBAPP_URL") || "";
    var repPasscode = props.getProperty("REP_PASSCODE") || "8899";
    var studentPasscode = props.getProperty("STUDENT_PASSCODE") || "1234";
    var adminUser = props.getProperty("ADMIN_USER") || "Abhi9av";
    var adminPass = props.getProperty("ADMIN_PASS") || "Abhi9av@1729";
    
    var logs = [];
    var startRow = Math.max(1, data.length - 100);
    
    for (var i = startRow; i < data.length; i++) {
      if (!data[i][2]) continue; // Skip empty rows
      
      var rawDate = data[i][0];
      var cleanDate = "";
      if (rawDate instanceof Date) {
        cleanDate = Utilities.formatDate(rawDate, Session.getScriptTimeZone(), "yyyy-MM-dd");
      } else {
        var str = String(rawDate).trim();
        var d = new Date(str);
        if (!isNaN(d.getTime())) {
          cleanDate = Utilities.formatDate(d, Session.getScriptTimeZone(), "yyyy-MM-dd");
        } else {
          cleanDate = str;
        }
      }
      
      logs.push({
        dateStr: cleanDate,
        timeStr: String(data[i][1] || ''),
        regNo: String(data[i][2] || ''),
        name: String(data[i][3] || ''),
        course: String(data[i][4] || ''),
        section: String(data[i][5] || '')
      });
    }
    
    var resultObj = {
      result: "success", 
      lastClearedAt: lastClearedAt,
      webAppUrl: webAppUrl,
      repPasscode: repPasscode,
      studentPasscode: studentPasscode,
      adminUser: adminUser,
      adminPass: adminPass,
      logs: logs
    };
    
    // Support JSONP for 100% bulletproof cross-origin fetching on all mobile browsers
    if (e && e.parameter && e.parameter.callback) {
      var cb = e.parameter.callback;
      return ContentService.createTextOutput(cb + '(' + JSON.stringify(resultObj) + ')')
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    
    return ContentService.createTextOutput(JSON.stringify(resultObj))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    if (e && e.parameter && e.parameter.callback) {
      return ContentService.createTextOutput(e.parameter.callback + '({"result":"error","logs":[]})')
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
    }
    return ContentService.createTextOutput(JSON.stringify({result: "error", logs: []}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
